function getRandomElement(arr) {
    if (!arr || arr.length === 0) return null;
    return arr[Math.floor(Math.random() * arr.length)];
}

function getRandomInt(min, max) {
    return Math.floor(Math.random() * (max - min + 1)) + min;
}

function generatePostalCode(countryCode, postalPrefix) {
    const code = countryCode.toUpperCase();
    
    if (code === 'US') {
        // US: 5-digit ZIP
        return postalPrefix + getRandomInt(10, 99);
    } else if (code === 'UK') {
        // UK: alphanumeric like SW1A 1AA
        const letters = 'ABDEFGHJLNPQRSTUWXYZ';
        return postalPrefix + getRandomInt(1, 9) + letters[getRandomInt(0, 19)] + ' ' + getRandomInt(1, 9) + letters[getRandomInt(0, 19)] + letters[getRandomInt(0, 19)];
    } else if (code === 'CA') {
        // CA: A1A 1A1 format
        const letters = 'ABCEGHJKLMNPRSTVXY';
        return postalPrefix + getRandomInt(0, 9) + letters[getRandomInt(0, 19)] + ' ' + getRandomInt(0, 9) + letters[getRandomInt(0, 19)] + getRandomInt(0, 9);
    } else if (code === 'AU') {
        // AU: 4-digit
        return postalPrefix + getRandomInt(0, 99).toString().padStart(2, '0');
    } else if (code === 'MM' || code === 'TH' || code === 'SG') {
        // MM/TH/SG: numeric
        return postalPrefix + getRandomInt(0, 999).toString().padStart(3, '0');
    }
    
    return postalPrefix + getRandomInt(100, 999);
}

function generatePhoneNumber(phoneCode, countryCode) {
    const code = countryCode.toUpperCase();
    
    if (code === 'US' || code === 'CA') {
        return `${phoneCode}${getRandomInt(200, 999)}${getRandomInt(200, 999)}${getRandomInt(1000, 9999)}`;
    } else if (code === 'UK') {
        return `${phoneCode}7${getRandomInt(100, 999)}${getRandomInt(100000, 999999)}`;
    } else if (code === 'AU') {
        return `${phoneCode}4${getRandomInt(10000000, 99999999)}`;
    } else if (code === 'MM') {
        return `${phoneCode}9${getRandomInt(200000000, 999999999)}`;
    } else if (code === 'TH') {
        return `${phoneCode}${getRandomInt(80, 99)}${getRandomInt(1000000, 9999999)}`;
    } else if (code === 'SG') {
        return `${phoneCode}${getRandomInt(8000, 9999)}${getRandomInt(1000, 9999)}`;
    }
    
    return `${phoneCode}${getRandomInt(100000000, 999999999)}`;
}

function generatePassword() {
    const chars = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789!@#$%^&*()_+";
    let password = "";
    password += "ABCDEFGHIJKLMNOPQRSTUVWXYZ"[getRandomInt(0, 25)];
    password += "abcdefghijklmnopqrstuvwxyz"[getRandomInt(0, 25)];
    password += "0123456789"[getRandomInt(0, 9)];
    password += "!@#$%^&*()_+"[getRandomInt(0, 11)];
    for (let i = 0; i < 8; i++) {
        password += chars[getRandomInt(0, chars.length - 1)];
    }
    return password.split('').sort(() => 0.5 - Math.random()).join('');
}

function generateEmail(fullName, countryCode) {
    const cleanName = fullName.toLowerCase().replace(/\s+/g, '.');
    const domains = ['gmail.com', 'yahoo.com', 'outlook.com', 'hotmail.com', 'mail.com'];
    return `${cleanName}${getRandomInt(1, 999)}@${getRandomElement(domains)}`;
}

function generateDateOfBirth() {
    const year = getRandomInt(1960, 2005);
    const month = getRandomInt(1, 12).toString().padStart(2, '0');
    const day = getRandomInt(1, 28).toString().padStart(2, '0');
    return `${year}-${month}-${day}`;
}

function generateAddress(countryCode) {
    const code = countryCode.toUpperCase();
    const country = COUNTRY_DATA[code];
    
    if (!country) {
        return null;
    }
    
    const cityData = getRandomElement(country.cities);
    const streetName = getRandomElement(cityData.streets);
    const streetNumber = getRandomInt(1, 999);
    const postalCode = generatePostalCode(code, cityData.postalPrefix);
    const phoneNumber = generatePhoneNumber(country.phoneCode, code);
    const fullName = generateFullName(code);
    const gender = Math.random() < 0.5 ? 'Male' : 'Female';
    const email = generateEmail(fullName, code);
    const dob = generateDateOfBirth();
    const password = generatePassword();
    
    return {
        person_name: fullName,
        gender: gender,
        street_address: `${streetNumber} ${streetName}`,
        city: cityData.city,
        state: cityData.state,
        postal_code: postalCode,
        phone_number: phoneNumber,
        email: email,
        date_of_birth: dob,
        password: password,
        country: country.name,
        country_code: code,
        flag: country.flag,
        currency: country.currency
    };
}

// ============================================================
// REQUEST HANDLER
// ============================================================

export default {
    async fetch(request, env, ctx) {
        // CORS headers
        const corsHeaders = {
            'Access-Control-Allow-Origin': '*',
            'Access-Control-Allow-Methods': 'GET, OPTIONS',
            'Access-Control-Allow-Headers': 'Content-Type',
        };
        
        // Handle OPTIONS preflight
        if (request.method === 'OPTIONS') {
            return new Response(null, { headers: corsHeaders });
        }
        
        // Only allow GET
        if (request.method !== 'GET') {
            return new Response(JSON.stringify({ 
                error: 'Method not allowed. Use GET.' 
            }), {
                status: 405,
                headers: { ...corsHeaders, 'Content-Type': 'application/json' }
            });
        }
        
        const url = new URL(request.url);
        const path = url.pathname;
        
        // Health check endpoint
        if (path === '/' || path === '/health') {
            return new Response(JSON.stringify({
                status: 'ok',
                service: 'Fake Address Generator API',
                version: '1.0.0',
                supported_countries: Object.keys(COUNTRY_DATA),
                endpoints: {
                    address: '/api/address?code=US',
                    countries: '/api/countries'
                }
            }), {
                headers: { ...corsHeaders, 'Content-Type': 'application/json' }
            });
        }
        
        // List supported countries
        if (path === '/api/countries') {
            const countries = Object.entries(COUNTRY_DATA).map(([code, data]) => ({
                code: code,
                name: data.name,
                flag: data.flag,
                currency: data.currency
            }));
            return new Response(JSON.stringify({
                total: countries.length,
                countries: countries
            }, null, 2), {
                headers: { ...corsHeaders, 'Content-Type': 'application/json' }
            });
        }
        
        // Main address generation endpoint
        if (path === '/api/address') {
            const countryCode = url.searchParams.get('code');
            
            if (!countryCode) {
                return new Response(JSON.stringify({
                    error: 'Missing parameter: code',
                    example: '/api/address?code=US',
                    supported: Object.keys(COUNTRY_DATA)
                }), {
                    status: 400,
                    headers: { ...corsHeaders, 'Content-Type': 'application/json' }
                });
            }
            
            const code = countryCode.toUpperCase();
            
            if (!COUNTRY_DATA[code]) {
                return new Response(JSON.stringify({
                    error: `Country code '${code}' is not supported`,
                    supported: Object.keys(COUNTRY_DATA)
                }), {
                    status: 404,
                    headers: { ...corsHeaders, 'Content-Type': 'application/json' }
                });
            }
            
            const address = generateAddress(code);
            
            return new Response(JSON.stringify({
                success: true,
                data: address
            }, null, 2), {
                headers: { ...corsHeaders, 'Content-Type': 'application/json' }
            });
        }
        
        // 404 for unknown paths
        return new Response(JSON.stringify({
            error: 'Not found',
            available_endpoints: ['/', '/health', '/api/countries', '/api/address?code=XX']
        }), {
            status: 404,
            headers: { ...corsHeaders, 'Content-Type': 'application/json' }
        });
    }
};
